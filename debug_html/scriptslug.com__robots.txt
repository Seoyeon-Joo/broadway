# ==============================================
# Yandex - Clean-param for URL parameters that do not change page content
# https://yandex.ru/support/webmaster/en/robot-workings/clean-param
# Yandex ignores the "User-agent: *" group once it finds its own group,
# so the general rules are repeated here. "/*?q=" is intentionally not
# disallowed for Yandex: Clean-param folds those URLs into the base page
# instead of dropping them (Yandex's recommended approach).
# ==============================================

User-agent: Yandex
Disallow: /craft/
Disallow: /request/
Disallow: /cpresources/
Disallow: /api/
Allow: /ads.txt
# Campaign tracking appended to links (newsletters use utm_source/utm_medium/utm_campaign)
Clean-param: utm_source&utm_medium&utm_campaign&utm_term&utm_content
# Click IDs appended by ad and social platforms when links are shared
Clean-param: gclid&fbclid
# Site search: the search overlay writes ?q= onto whatever page it is opened from,
# /scripts/series?q= filters the series list, and newsletters link to /contact?q=
Clean-param: q
# Listing sort order on the home page and /scripts/* listing pages (?sort=).
# ?pg= is deliberately NOT listed: it changes which items are on the page.
Clean-param: sort
# Contact form state: ?subject= prefills the form, ?s= is the post-submit status
Clean-param: subject&s /contact
# PDF viewer deep link (?page=N) on script pages; the HTML page is identical
Clean-param: page /script/
# Image cache-buster appended to poster/backdrop/transform URLs
Clean-param: utime
